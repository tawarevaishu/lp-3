pragma solidity ^0.8.20;

contract StudentData {

    // 1. Structure to store student information
    struct Student {
        uint rollNo;
        string name;
        string department;
        uint marks;
    }

    // 2. Array to store multiple students
    Student[] public students;

    // Event to record student addition
    event StudentAdded(uint rollNo, string name);

    // Event to record fallback execution
    event FallbackExecuted(address sender, uint value);

    // 3. Function to add student details
    function addStudent(
        uint _rollNo,
        string memory _name,
        string memory _department,
        uint _marks
    ) public {

        students.push(
            Student(_rollNo, _name, _department, _marks)
        );

        emit StudentAdded(_rollNo, _name);
    }

    // Function to retrieve student details
    function getStudent(uint index)
        public
        view
        returns (
            uint,
            string memory,
            string memory,
            uint
        )
    {
        require(index < students.length, "Invalid index");

        Student memory s = students[index];

        return (
            s.rollNo,
            s.name,
            s.department,
            s.marks
        );
    }

    // Function to display total students
    function getStudentCount() public view returns (uint) {
        return students.length;
    }

    // Function to test fallback through an unmatched call
    function triggerFallback() public {
        (bool success, ) = address(this).call(hex"12345678");
        require(success, "Fallback call failed");
    }

    // Handles empty-calldata Ether transfers
    receive() external payable {}

    // 4. Fallback function
    fallback() external payable {
        emit FallbackExecuted(msg.sender, msg.value);
    }
}
